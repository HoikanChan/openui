root = Stack([header, tableCard])
header = CardHeader("CPU利用率 Top 3 设备", "内存利用率详情")
tableCard = Card([top3Table])
top3Rows = @Filter(data.queryNetworkDeviceMetrics.telemetryData, "cpu_usage", ">=", 0)
sortedRows = @Sort(top3Rows, "cpu_usage", "desc")
top3Data = @Each(sortedRows, "item", @MergeObject(item, {rank: @Count(@Filter(sortedRows, "cpu_usage", ">", item.cpu_usage)) + 1}))
limitedData = @Filter(top3Data, "rank", "<=", 3)
top3Table = Table([nameCol, ipCol, cpuCol, memCol], limitedData)
nameCol = Col("设备名称", "neName")
ipCol = Col("设备IP", "neIp")
cpuCol = Col("CPU利用率 (%)", "cpu_usage")
memCol = Col("内存利用率 (%)", "memory_usage", {cell: @Render("v", TextContent("N/A"))})
