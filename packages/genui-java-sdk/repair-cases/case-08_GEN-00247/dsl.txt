root = Stack([headerCard, moduleTabs], "column")

// Group data by optical module (opt_uniq_id)
modules = @ObjectValues(@GroupBy(data.queryOpticalModuleMetrics.telemetryData, "opt_uniq_id"))
moduleTabs = Tabs(@Each(modules, "modGroup", TabItem(modGroup[0].opt_uniq_id, modGroup[0].ifName, [moduleSummaryCard, laneTable])))

moduleSummaryCard = Card([
  CardHeader("光模块维度指标", modGroup[0].opt_uniq_id),
  Stack([tempCard, voltCard], "row")
])
tempCard = Card([TextContent("温度", "small"), TextContent("N/A ℃", "large-heavy")])
voltCard = Card([TextContent("电压", "small"), TextContent("N/A V", "large-heavy")])

laneTable = Table([
  Col("通道号", "laneId"),
  Col("发送功率 (dBm)", "tx_power_dbm"),
  Col("发送功率(过滤) (dBm)", "tx_power_dbm_filter"),
  Col("接收功率 (dBm)", "rx_power_dbm"),
  Col("接收功率(过滤) (dBm)", "rx_power_dbm_filter"),
  Col("电流 (mA)", "tx_bias_ma")
], modGroup)
