root = Stack([summaryCard, typeDistCard, deviceTableCard])
summaryCard = Card([summaryHeader, summaryContent])
summaryHeader = CardHeader("Fabric 设备概览", "Fabric: default1")
summaryContent = Stack([totalKpi, sceneKpi], "row")
totalKpi = Card([TextContent("设备总数", "small"), TextContent("" + data.queryDeviceInfo.deviceTotal, "large-heavy")])
sceneKpi = Card([TextContent("部署场景", "small"), TextContent(data.queryDeviceInfo.deployScene, "large-heavy")])

typeDistCard = Card([typeDistHeader, typeDistContent])
typeDistHeader = CardHeader("款型分布", "按 neType 分组")
typeDistContent = Table([typeCol, countCol], typeDistRows)
typeCol = Col("款型 (neType)", "type")
countCol = Col("数量", "count")
typeDistRows = @Each(@ObjectEntries(typeCounts), "entry", @MergeObject({type: entry.key}, {count: entry.value}))
typeCounts = @ObjectEntries(@GroupBy(data.queryDeviceInfo.deviceInfo, "neType"))

deviceTableCard = Card([deviceTableHeader, deviceTable])
deviceTableHeader = CardHeader("设备列表", "共 " + data.queryDeviceInfo.deviceTotal + " 台设备")
deviceTable = Table([indexCol, nameCol, ipCol, typeCol2, stateCol, roleCol, vendorCol], data.queryDeviceInfo.deviceInfo)
indexCol = IndexCol("序号")
nameCol = Col("设备名称", "neName")
ipCol = Col("IP 地址", "neIp")
typeCol2 = Col("款型", "neType")
stateCol = Col("状态", "neState", {cell: @Render("v", @Switch(v, {"online": Tag("在线", "success"), "offline": Tag("离线", "danger"), "unknown": Tag("未知", "neutral")}, Tag(v)))})
roleCol = Col("角色", "neRole", {cell: @Render("v", @Switch(v, {"ServerLeaf": "Server Leaf"}, v))})
vendorCol = Col("厂商", "neVendor")
